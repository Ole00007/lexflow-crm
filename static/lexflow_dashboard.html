📊 API Endpoints Map

# AUTH
  POST /api/auth/login        → get JWT token
  GET  /api/auth/me           → current user info

# CONTACTS (open GET, auth for write)
  GET  /api/contacts          → list all
  GET  /api/contacts/:id      → get one
  POST /api/contacts          → create
  PUT  /api/contacts/:id      → update
  DELETE /api/contacts/:id    → soft delete

# CASES (open GET, auth for write)
  GET  /api/cases             → list all
  GET  /api/cases/:id         → get one
  POST /api/cases             → create
  PUT  /api/cases/:id         → update
  DELETE /api/cases/:id       → soft delete

# NOTES (JWT required everywhere)
  GET  /api/notes?target_type=case&target_id=1
  POST /api/notes
  PUT  /api/notes/:id
  DELETE /api/notes/:id

# ACTIVITY (open)
  GET  /api/activity?target_type=case&target_id=1
  GET  /api/activity          → all timeline

# CALENDAR (JWT required)
  GET  /api/calendar?caseid=1&event_type=hearing
  POST /api/calendar
  PUT  /api/calendar/:id
  DELETE /api/calendar/:id

# DEADLINES (JWT required)
  GET  /api/deadlines?caseid=1
  POST /api/deadlines
  PUT  /api/deadlines/:id
  DELETE /api/deadlines/:id

# ADMIN (JWT + admin role)
  GET  /api/admin/health
  GET  /api/admin/stats
  GET  /api/admin/users
  PUT  /api/admin/users/:id/role
  DELETE /api/admin/users/:id

# TASKS (JWT required)
  GET  /api/tasks
  POST /api/tasks
  PUT  /api/tasks/:id
  DELETE /api/tasks/:id

# WEBHOOKS (chatbot integration)
  POST /api/webhooks/chatbot/message
